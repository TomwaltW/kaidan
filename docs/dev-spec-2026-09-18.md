## 1. 问题定义

**要做什么**（一句话）：
搭出开单 MVP——一个 pnpm + TypeScript monorepo，让销售在飞书（或本地 CLI）里通过 7 个技能读取 Mock / 飞书多维表格 CRM、获得晨报 / 会前 / 商机体检 / 周报，并以"提案 → 人工批准 → 提交"的方式更新 CRM；同时交付 MCP server、虚构评测集与端到端验收用例。

**为什么要做**：
验证"Claudeforce 模式（CRM 以技能形式进入 Agent）"在中国中小 B2B 销售团队中的需求。第 2 周要拿它做客户演示与付费试点，所以三件事必须过硬：演示路径（Mock 数据 + CLI + 飞书）稳定；安全承诺真实（模型只提案不提交、越权为 0、手机号只出脱敏值）；换模型、换客户只改配置。模糊地带一律往"更安全、更可测、更少依赖"偏。

**明确不做**：
- 不做 CRM 本体、不做通用 Agent 平台、不做多租户（一个实例 = 一个客户）
- 不接任何境外模型（Claude / OpenAI 等），不做"模型中转"
- 不做飞书用户授权（user_access_token）模式、飞书日历、飞书妙记、群聊摄取（均为 P1）
- 不做钉钉 / 企微入口、纷享销客适配器、生成式 HTML 看板
- 不引入 契约之外的新依赖；不引入任何需要原生编译的依赖（SQLite 用 Node 内置 `node:sqlite`）
- 不写生产部署自动化（只要求 Dockerfile 与 CI 文件存在且合理）

```ts
/** Mock 适配器与默认展示用的中文阶段名 */
export const DEFAULT_STAGE_LABELS: Readonly<Record<OpportunityStage, string>> = {

  solution: "方案报价",

  won: "赢单",
```

`packages/contracts/src/adapter.ts`

```ts
// @kaidan/contracts — adapter.ts（冻结）
import { z } from "zod";
import { Actor, RomObjectType, RomRecord, RomRef, type RomDataset } from "./rom.js";

export const FilterOp = z.enum(["eq", "neq", "lt", "lte", "gt", "gte", "in", "contains", "is_empty", "not_empty"]);
export const Filter = z.object({ field: z.string().min(1), op: FilterOp, value: z.unknown().optional() });
export type Filter = z.infer<typeof Filter>;

export const RomQuery = z.object({
  type: RomObjectType,
  filters: z.array(Filter).max(10).optional(),
  /** 在 name / subject / title / summary 上做包含匹配 */
  text: z.string().max(100).optional(),
  sort: z.object({ field: z.string().min(1), dir: z.enum(["asc", "desc"]) }).optional(),
  limit: z.number().int().min(1).max(100).optional(), // 缺省 20
  cursor: z.string().optional(),
});
export type RomQuery = z.infer<typeof RomQuery>;

export interface RomPage {
  items: RomRecord[];
  nextCursor?: string;
}

export interface WriteOptions {
  /** = proposalId；同一 key 重复写入必须幂等（不产生第二次写） */
  idempotencyKey: string;
  proposalId: string;
}

export const SchemaInfo = z.object({
  source: z.enum(["mock", "bitable"]),
  tables: z.array(
    z.object({
      id: z.string(),
      name: z.string(),
      fields: z.array(z.object({ name: z.string(), kind: z.string(), options: z.array(z.string()).optional() })),
    }),
  ),
});
export type SchemaInfo = z.infer<typeof SchemaInfo>;

/**
 * 可见性规则 V1（在"适配器自行鉴权"的模式下必须实现：mock、bitable tenant 模式）
 *  - admin：全部可见
 *  - manager：ownerId ∈ {自己} ∪ teamIds
 *  - rep：ownerId = 自己
 *  - contact / activity / task 额外规则：若其 accountId 或 opportunityId 指向的记录可见，则它也可见
 * 不可见记录：search 结果中剔除；get/update 抛 KaidanError(PERMISSION_DENIED)
 * 写入：只能改可见记录；create 的 ownerId 强制为 actor.userId
 */
export interface RomAdapter {
  readonly kind: "mock" | "bitable";
  search(actor: Actor, q: RomQuery): Promise<RomPage>;
  /** 不存在 → 返回 null；存在但不可见 → 抛 PERMISSION_DENIED */
  get(actor: Actor, ref: RomRef): Promise<RomRecord | null>;
  create(actor: Actor, type: RomObjectType, data: Record<string, unknown>, opts: WriteOptions): Promise<RomRecord>;
  update(actor: Actor, ref: RomRef, patch: Record<string, unknown>, opts: WriteOptions): Promise<RomRecord>;
  /** 仅 admin；否则 PERMISSION_DENIED */
  inspectSchema(actor: Actor): Promise<SchemaInfo>;
}

export interface MockRomAdapter extends RomAdapter {
  readonly kind: "mock";
  /** writes = 实际发生的写入次数（幂等命中不计） */
  stats(): { writes: number; appliedKeys: string[] };
```

```ts
    })

export type KaidanConfig = z.infer<typeof KaidanConfig>;

/**
 * 读取 YAML 配置：
 *  - zod 校验失败 → KaidanError(CONFIG)，message 含字段路径
 *  - 相对路径字段（adapter.mock.datasetPath / adapter.bitable.mappingPath / governance.dbPath（非 :memory:）/ skillsDir）
 *    解析为相对于配置文件所在目录的绝对路径
 *  - adapter.kind 与对应子配置必须同时存在，否则 CONFIG
 */
export async function loadConfig(path: string): Promise<KaidanConfig> {
  const text = await readFile(path, "utf8");
  const parsed = KaidanConfig.safeParse(parse(text));
  if (!parsed.success) {
    const where = parsed.error.issues.map((i) => `${i.path.join(".")}: ${i.message}`).join("; ");
    throw new KaidanError(ErrorCode.CONFIG, `配置无效：${where}`);
  }
  const cfg = parsed.data;
  if (cfg.adapter.kind === "mock" && !cfg.adapter.mock) throw new KaidanError(ErrorCode.CONFIG, "adapter.mock 缺失");
  if (cfg.adapter.kind === "bitable" && !cfg.adapter.bitable)
    throw new KaidanError(ErrorCode.CONFIG, "adapter.bitable 缺失");
  const base = dirname(resolve(path));
  const abs = (p: string) => (isAbsolute(p) ? p : resolve(base, p));
  if (cfg.adapter.mock) cfg.adapter.mock.datasetPath = abs(cfg.adapter.mock.datasetPath);
  if (cfg.adapter.bitable) cfg.adapter.bitable.mappingPath = abs(cfg.adapter.bitable.mappingPath);
  if (cfg.governance.dbPath !== ":memory:") cfg.governance.dbPath = abs(cfg.governance.dbPath);
  if (cfg.skillsDir) cfg.skillsDir = abs(cfg.skillsDir);
  return cfg;
}
```

```ts
 *  - reply 卡片页脚格式："内容由 AI 生成 · {providerName} {modelName}" + （有 filingNo 时）" · 备案号 {filingNo}"
 *  - handleCardAction：调用 approvals.approve/reject；成功返回 notice 卡片（approve 文案含"已写入"，reject 含"已拒绝"）；
 *    KaidanError 返回 notice 卡片说明原因（PERMISSION_DENIED→"你无权审批该提案"，EXPIRED→"提案已过期"，CONFLICT→"记录已被他人修改，请重新生成"）
 *  - runScheduled：对 identity.listScheduled() 的每个用户以 channel="schedule" 运行对应技能（daily_brief→daily-brief，weekly_review→weekly-review），
 *    私聊发送 reply 卡片；单个用户失败不影响其他用户
 */
export interface SurfaceHandlers {
  handleMessage(msg: InboundMessage): Promise<void>;
  handleCardAction(a: CardAction): Promise<OutboundCard>;
  runScheduled(kind: "daily_brief" | "weekly_review"): Promise<{ sent: number; failed: number }>;
}

export interface Surface extends SurfaceHandlers {
  /** 建立飞书长连接 + 启动定时任务 */
  start(): Promise<void>;
  stop(): Promise<void>;
}

export interface SurfaceDeps {
  agent: AgentRuntime;
  governance: Governance;
  identity: IdentityResolver;
  config: KaidanConfig;
  client?: LarkClientLike; // 测试注入 fake；缺省用 @larksuiteoapi/node-sdk 实现
}
```

**跨 track 依赖的行为约定**（契约注释之外的补充，同样冻结）：

1. **审计事件由谁写、data 放什么**：`skill_invoked`（data `{skill, channel}`）、`llm_call`（data `{provider, inputTokens, outputTokens, finishReason}`）→ agent；`tool_call`（data `{tool, ok, ms, errorCode?}`）、`permission_denied`（data `{tool}`）→ tools（ToolRegistry.invoke）；`proposal_created`（data `{proposalId, objectType, kind}`）→ `ProposalService.create`；`proposal_committed / rejected / failed / expired`（data `{proposalId, by?, error?}`）→ `ApprovalService`。agent 在技能 roles 不允许时写 `permission_denied`（data `{skill}`）。
2. **Mock 适配器**：`search` 的 `text` 匹配 name / subject / title / summary（包含即可）；`filters` 语义按字段值比较（日期字符串按字典序）；`sort` 缺省按 `id` 升序；`cursor` 为十进制偏移量字符串；`limit` 缺省 20。`update` 改了 `stage` 时同步 `rawStage = DEFAULT_STAGE_LABELS[stage]`；写入时 `updatedAt` 取真实当前时间（测试不对其断言）；`create` 的 id 为 `<type>_new_<自增序号>`，`ownerId` 强制为 actor；写入前把合并后的记录用对应 ROM schema 校验，不合法 → `VALIDATION`；同一 `idempotencyKey` 第二次写直接返回当前记录且不计入 `writes`。
3. **工具**：`propose_update` 先 `adapter.get(actor, ref)`（权限在此检查），只保留与当前值不同的字段进 diff（顺序 = patch 的键顺序），记录中不存在的字段 `before` 记为 `null`，diff 为空 → `VALIDATION("没有变化")`；合并结果须通过 ROM schema 校验。`propose_create` 的 diff = 每个字段 `{before:null, after:值}`，引用的 accountId / opportunityId 必须对 actor 可见。`schema_inspect` 在工具层先检查 admin。`pipeline_snapshot` 的"今天"按 `deps.timezone` 与 `deps.clock` 计算，通过 `adapter.search` 翻页取全量（每页 100）。
4. **agent**：系统提示 = `systemPreamble`（可选）+ `SECURITY_PREAMBLE` + 技能正文 + "当前时间（timezone）与当前用户（displayName、roles）"；用户消息 = 去掉命令后的文本（为空时用技能名）；工具规格 = `registry.list({names: skill.allowedTools})` 映射为 `{name, description, parameters: z.toJSONSchema(input) 去掉顶层 $schema}`；同一步多个工具调用按顺序执行；`reply` = 最后一条 assistant 的 content（null 视为 ""）；`proposals` = 本次成功的 `propose_*` 输出里的 proposalId 经 `proposals.get` 取回。
5. **OpenAI 兼容 provider**：用 Node 内置 `fetch` 调 `POST {baseURL}/chat/completions`（不引入任何 SDK）；构造时不读密钥、不联网；首次 `chat` 时读 `process.env[apiKeyEnv]`，缺失 → `CONFIG`；HTTP 429 → `RATE_LIMITED`（retryable），5xx / 网络错误 / 60 秒超时 → `UPSTREAM`（retryable），其他 4xx → `UPSTREAM`（不可重试）；`id` = `"openai_compatible:" + model`。
6. **脚本 provider**：第 i 次 `chat` 返回第 i 步；有 `toolCalls` 时 assistant 消息带 `toolCalls`（id 为 `call_<i>_<j>`，arguments 为 `JSON.stringify`）且 `finishReason="tool_calls"`，否则 `content` + `"stop"`；usage 恒为 0；`requests` 保存每次请求的深拷贝；`id="scripted"`；缺省 disclosure `{providerName:"脚本", modelName:"scripted"}`。
7. **loadSkills**：读取 `dir` 下每个子目录里的 `SKILL.md`（没有 SKILL.md 的子目录跳过），frontmatter 用 `---` 包裹的 YAML，按 `SkillFrontmatter` 校验，`allowed_tools` 映射为 `allowedTools`，按 `name` 排序；不合法 → `CONFIG`（message 含文件路径）。
8. **飞书入口**：`createFeishuSurface` 构造时不联网、不读密钥；只有 `start()` 或在没有注入 client 时首次发送才需要 `config.feishu`，缺失 → `CONFIG`。身份解析：`Actor.userId = openId`，`tenantId = config.tenantId`，manager 的 `teamIds` = 所有 `managerOpenId` 等于他的用户（按 config 顺序）；`listScheduled()` 按 config 顺序。
9. **Governance**：用 `node:sqlite` 的 `DatabaseSync`；`dbPath` 不是 `:memory:` 且父目录不存在时自动递归创建；提案 id 前缀 `prop_`、审计 id 前缀 `aud_`（后接 `crypto.randomUUID()`）；时间一律取 `deps.clock`；`close()` 关闭数据库。

### 3.3 失败面

| 场景 | 行为 |
|---|---|
| 工具输入不合法 / 未知工具 / 技能不允许的工具 | `ToolRegistry.invoke` 返回 `ok:false, code:VALIDATION`；agent 把它作为工具消息交给模型，不中断 |
| 读取不可见记录 | adapter 抛 `PERMISSION_DENIED`；invoke 返回 `ok:false`，message 固定为"无权限访问该记录"，不带任何记录内容；写 permission_denied |
| 记录不存在 | `get` 返回 `null`；`crm_get` 输出 `{item:null}`；`propose_update` → `NOT_FOUND` |
| 非 admin 调 schema_inspect / rep 调 team 视角 | `PERMISSION_DENIED` |
| 提案审批人无权 | `approve/reject` 抛 `PERMISSION_DENIED`，提案状态不变 |
| 提案过期 | 首次访问抛 `EXPIRED` 并置 `expired`；之后抛 `INVALID_STATE` |
| 提交时记录已被改 | 抛 `CONFLICT`，提案置 `failed`，`error` 写原因，不写入 |
| 适配器写入失败 | 提案置 `failed`，原错误继续抛出 |
| 模型接口 429 / 5xx / 超时 | `RATE_LIMITED` / `UPSTREAM`（retryable=true）；agent 不重试，`stoppedBy="error"`，reply 为友好提示，不抛出 |
| 超过 maxSteps / maxToolCalls / maxInputTokens | agent 停止，`stoppedBy="max_steps"` / `"budget"`，reply 说明原因 |
| 技能 roles 不允许 | 不调模型，`stoppedBy="forbidden"`，写 permission_denied |
| 配置无效 / 缺密钥 / 技能文件不合法 | `CONFIG`，message 指明字段或文件 |
| 飞书 API 频控 | 适配器与入口各自最多重试 3 次（指数退避 0.5s / 1s / 2s），仍失败 → `RATE_LIMITED` |
| 调用未实现的桩 | `NOT_IMPLEMENTED` |

## 7. 卡住了怎么办

按顺序：

1. 需要的接口在契约里找不到 → **停，报告**。不要自己发明一个。
2. 需要改的文件不在自己的白名单里 → **停，报告**。不要"就改一行"。
3. 需要新依赖 → **停，报告**。依赖由 T0 冻结。
4. 验收命令本身跑不起来（环境问题）→ 先自己排查环境，排查不动 → 报告。
5. 验收过不了但代码逻辑自认为对 → **以验收为准**，继续改；确信是验收标准写错了 → 停，报告。
6. 飞书 / MCP SDK 的接口拿不准 → 以 `node_modules` 中安装版本的类型定义为准；仍拿不准就把依赖点封装在一个小接口后面，用 fake 测，并在回执里列出待真机验证的点。

报告 = 在回执里写清楚卡在哪、试过什么、需要什么决定，然后结束这一轮，不要继续硬做。
